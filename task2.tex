\section*{Continuity Problems}
% ------------------------------------------------

\begin{enumerate}

\item Prove $f(x)=5x-3$ is continuous at $x=0,-3,5$.

\item Examine continuity of $f(x)=2x^2-1$ at $x=3$.

\item Examine continuity:
\begin{enumerate}
\item $f(x)=x-5$
\item $f(x)=\dfrac{1}{x-5},\; x\neq5$
\item $f(x)=\dfrac{x^2-25}{x+5},\; x\neq-5$
\item $f(x)=|x-5|$
\end{enumerate}

\item Prove $f(x)=x^n$ is continuous.

\item
\[
f(x)=
\begin{cases}
x,& x\le1\\
5,& x>1
\end{cases}
\]
Check continuity at $0,1,2$.

\item
\[
f(x)=
\begin{cases}
x+3,& x\le2\\
x-3,& x>2
\end{cases}
\]

\item
\[
f(x)=
\begin{cases}
|x|+3,& x\le-3\\
2x,& -3<x<3\\
6x-2,& x\ge3
\end{cases}
\]

\item
\[
f(x)=
\begin{cases}
|x|,& x\neq0\\
0,& x=0
\end{cases}
\]

\item
\[
f(x)=
\begin{cases}
|x|,& x<0\\
1,& x\ge0
\end{cases}
\]

\item
\[
f(x)=
\begin{cases}
x^2+1,& x\ge1\\
x+1,& x<1
\end{cases}
\]

\item Show $g(x)=x-[x]$ is discontinuous at integers.

\item Check continuity of $f(x)=x^2-\sin x+5$ at $x=\pi$.

\item Discuss continuity of $\sin x+\cos x$, $\sin x-\cos x$, $\sin x\cos x$.

\item Discuss continuity of cosine, cosecant, secant and cotangent.

\item Determine continuity of
\[
f(x)=
\begin{cases}
\dfrac{\sin x}{x},&x\neq0\\
0,&x=0
\end{cases}
\]

\item Show $f(x)=\cos(x^2)$ is continuous.

\item Show $f(x)=|\cos x|$ is continuous.

\item Show $\sin|x|$ is continuous.

\item Find discontinuities of $f(x)=|x|-|x+1|$.

\end{enumerate}

\section*{Differentiation}
% ------------------------------------------------

\begin{enumerate}
\item $\sin(x^2+5)$
\item $\cos(\sin x)$
\item $\sin(ax+b)$
\item $\sec(\tan x)$
\item $\dfrac{\sin(ax+b)}{\cos(cx+d)}$
\item $\cos(x^3)\sin^2(x^5)$
\item $(2\cot x)^2$
\item $\cos(\sqrt{x})$
\end{enumerate}

% ------------------------------------------------
\section*{Non-Differentiability}
% ------------------------------------------------

\begin{enumerate}
\item Prove $|x-1|$ is not differentiable at $x=1$.
\item Prove greatest integer function $[x]$ is not differentiable at $x=1$ and $x=2$.
\end{enumerate}

% ------------------------------------------------
\section*{Implicit Differentiation}
% ------------------------------------------------

Find $\dfrac{dy}{dx}$:

\begin{enumerate}
\item $2x+3y=\sin x$
\item $2x+3y=\sin y$
\item $ax+by^2=\cos y$
\item $xy+y^2=\tan x+y$
\item $x^2+xy+y^2=100$
\item $x^3+x^2y+xy^2+y^3=81$
\item $\sin^2 y+\cos(xy)=k$
\item $\sin^2 x+\cos^2 y=1$
\end{enumerate}
